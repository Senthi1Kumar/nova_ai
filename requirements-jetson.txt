# Jetson / aarch64 overrides — install AFTER uv sync
#
# The standard PyPI torch (torch==2.10.0+cu130) installed by uv sync is built
# for x86_64 and has cuBLAS incompatibilities on Jetson (aarch64), causing
# CUBLAS_STATUS_INVALID_VALUE crashes in LFM2.5-350M's rotary embeddings.
#
# NVIDIA ships Jetson-specific torch wheels via the Jetson AI Lab index.
# Run this file with the .venv pip AFTER uv sync:
#
#   .venv/bin/pip install -r requirements-jetson.txt --extra-index-url https://pypi.jetson-ai-lab.dev/simple/
#
# This replaces only the CUDA compute stack — all other deps from uv sync remain.

--extra-index-url https://pypi.jetson-ai-lab.dev/simple/

# Core CUDA compute stack — Jetson-native builds
torch
torchaudio
torchvision

# onnxruntime — GPU build for Jetson (replaces CPU-only onnxruntime from uv sync)
# Provides CUDA acceleration for Moonshine STT and Gemma intent classifier
onnxruntime-gpu
